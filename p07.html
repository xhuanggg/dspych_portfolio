This portflio piece does different kinds of automated bivariate association tests for an exploratory project. Being automated means there is no need to running the tests one by one. Instead, it can return the test results with Bonferroni correction and export the results in Excel sheets just by inputting the IVs and DV.

This piece runs Pearson’s r (continuous IV and DV), ANOVA (categorical IV and continuous DV), and chi-square test (categorical IV and DV).

  y <- df_anova$mCO1_ImpactOnRelationship

  # Perform ANOVA
  model <- lm(y ~ x, data=df)  
  anova_result <- Anova(model, type="III")
  
  # Print degrees of freedom
  df1 <- anova_result$`Df`[2]
  df1
  df2 <- anova_result$`Df`[3]
  df2
  
  # Print F-value
  f <- round(anova_result$`F value`[2], 2)
  f
  
  # Print p-value
  p <- ifelse(anova_result$`Pr(>F)`[2] > .999, 
                                ">.999", 
                                ifelse(anova_result$`Pr(>F)`[2] <.001, 
                                       "<.001", 
                                        round(anova_result$`Pr(>F)`[2], 3)
                                 )
   )
  p
  
  # Perform Bonferroni correction
  adjusted_p <- anova_result$`Pr(>F)`[2] * (3 * 85)
  adjusted_p <- ifelse(adjusted_p > .999, 
                                ">.999", 
                                ifelse(adjusted_p <.001, 
                                        "<.001", 
                                        round(adjusted_p, 3)
                                 )
  )
  adjusted_p
  
  # Calculate omega squared
  model <- aov(y ~ x, data=df)  
  SS_b <- summary(model)[[1]]$`Sum Sq`[1]  # Sum of squares between groups
  df_b <- summary(model)[[1]]$Df[1]       # Degrees of freedom between groups
  SS_w <- summary(model)[[1]]$`Sum Sq`[2]  # Sum of squares within groups
  df_w <- summary(model)[[1]]$Df[2]       # Degrees of freedom within groups
  MS_w <- SS_w / df_w                     # Mean square within groups
  omega2 <- (SS_b - df_b * MS_w) / (SS_b + SS_w + MS_w)
  omega2 <- round(omega2, 3)

  # Combine the row (two dfs and some values) to a blank df
  ANOVA.table_row <- cbind(var_i, df1, df2, f, p, adjusted_p, omega2)
  
  names(ANOVA.table_row) <- names(ANOVA.table)

  ANOVA.table <- rbind(ANOVA.table, ANOVA.table_row)
  glimpse(ANOVA.table) 
}

  print(paste("===============================", var_i, "==============================="))
  df <- df_chisq
  x <- df[[var_i]]
  y <- merged_650_r$mCO1_Reaction_LCA_class

  # Run Chi-Sq analysis
  # Use tryCatch to handle errors
result <- tryCatch(
  {
    CrossTable(table(x,y),expected=T,prop.r=F,prop.c=T,prop.t=F,prop.chisq=F,chisq=T,fisher=T,format="SPSS")
  },
  error = function(e) {
    # If an error occurs
    CrossTable(table(x,y),expected=T,prop.r=F,prop.c=T,prop.t=F,prop.chisq=F,chisq=T,simulate.p.value=TRUE, format="SPSS")
  }
)
  

  # Create a contingency table
  chi_sq_result <- table(x, y) %>%
    chisq.test()

  p_value <- ifelse(chi_sq_result$p.value > .999, 
                              ">.999", 
                              ifelse(chi_sq_result$p.value <.001, 
                                      "<.001", 
                                      round(chi_sq_result$p.value, 3)
                                )
  )

  # Conduct Bonferroni correction
  adjusted_p_value <- chi_sq_result$p.value * (3 * 85)
  adjusted_p_value <- ifelse(adjusted_p_value > .999, 
                              ">.999", 
                              ifelse(adjusted_p_value <.001, 
                                      "<.001", 
                                      round(adjusted_p_value, 3)
                                )
  )

  # Print the adjusted p-value
  print(adjusted_p_value)

  # Calculate Cramer's V
  tab1 <- gmodels::CrossTable(x,y)
  chisq_summary <- vcd::assocstats(tab1$t)
  chisq_value <- round(chisq_summary$chisq_tests["Pearson", "X^2"], 2)
  cramer_value <- round(chisq_summary$cramer, 2)

    # Combind the row (two dfs and some values) to a blank df
  chi.table_row <- cbind(var_i, chisq_value, p_value, adjusted_p_value, cramer_value)

  chi.table <- rbind(chi.table, chi.table_row)
