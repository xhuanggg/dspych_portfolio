library(dplyr)

process_all <- function(df, conds) {
  
  pattern <- paste0("^(", paste(conds, collapse="|"), ")_")
  vars <- names(df)
  
  items <- unique(sub(pattern, "", vars[grepl(pattern, vars)]))
  
  for (item in items) {
    
    cols <- paste0(conds, "_", item)
    cols <- cols[cols %in% names(df)]  # only keep existing ones
    
    # skip if less than 2 columns (safety)
    if (length(cols) == 0) next
    
    # v0
    df[[paste0(item, "_v0")]] <- coalesce(!!!df[cols])
    
    # C
    df[[paste0(item, "_C")]] <- conds[
      max.col(!is.na(df[cols]), ties.method = "first")
    ]
  }
  
  return(df)
}
